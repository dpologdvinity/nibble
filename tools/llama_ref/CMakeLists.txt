cmake_minimum_required(VERSION 3.25)
project(llama_ref CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(NIBBLE_GENERIC)
  # Generic ggml CPU path: no SIMD, no llamafile sgemm, no weight repack (see tools/make_golden.sh).
  foreach(o GGML_NATIVE GGML_AVX GGML_AVX2 GGML_AVX_VNNI GGML_FMA GGML_F16C GGML_AVX512 GGML_BMI2 GGML_LLAMAFILE GGML_CPU_REPACK)
    set(${o} OFF CACHE BOOL "" FORCE)
  endforeach()
else()
  set(GGML_NATIVE ON CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/llama.cpp llama.cpp)
add_executable(llama_ref llama_ref.cpp)
target_include_directories(llama_ref PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../tests)
target_link_libraries(llama_ref PRIVATE llama)
set_target_properties(llama_ref PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
